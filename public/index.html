🔌 API 文档（给开发者）

 
POST   /api/auth/register      { username, password } → { token }
POST   /api/auth/login         { username, password } → { token }
POST   /api/upload             FormData(file) + Authorization → { shareCode, url, expiresAt }
GET    /api/files              Authorization → [{ id, name, size, shareCode, downloadCount, expiresAt }]
DELETE /api/files/:id          Authorization → { success }
GET    /api/s/:code            公开下载（若设密码需先 POST /api/s/:code/verify 拿 token）
POST   /api/s/:code/verify     { password } → { token }
GET    /api/s/:code/page       公开分享预览页（HTML）
        
# 命令行调用示例
TOKEN=...   # 从 /api/auth/login 获取

# 1) 上传
curl -X POST http://localhost:3000/api/upload \
  -H "Authorization: Bearer $TOKEN" \
  -F "file=@./example.csv"

# 2) 列出我的文件
curl http://localhost:3000/api/files \
  -H "Authorization: Bearer $TOKEN"

# 3) 删除
curl -X DELETE http://localhost:3000/api/files/3 \
  -H "Authorization: Bearer $TOKEN"

# 4) 公开下载（无密码时）
curl -O http://localhost:3000/api/s/abc123
