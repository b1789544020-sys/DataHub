🔌 API 文档（给开发者）

 
POST   /api/auth/register      { username, password } → { token, user }
POST   /api/auth/login         { username, password } → { token, user }
GET    /api/auth/me            Authorization → { user }
POST   /api/upload             FormData(file[, password]) + Authorization
                               → { id, shareCode, shareUrl, expiresAt, ... }
GET    /api/files              Authorization → { files: [...], quota: { used, max } }
GET    /api/files/:id/meta     Authorization → 单个文件元数据
DELETE /api/files/:id          Authorization → { success }
GET    /api/s/:code            公开元数据（不下载）
POST   /api/s/:code/verify     { password } → { token }（仅加密文件需要）
GET    /api/s/:code/download   公开下载（加密文件带 ?token=...）
GET    /s/:code                公开分享页（HTML，浏览器直接打开）
        
# 命令行调用示例
TOKEN=...   # 从 /api/auth/login 获取

# 1) 上传（上传成功即返回短链接，无需二次调用）
curl -X POST http://localhost:3000/api/upload \
  -H "Authorization: Bearer $TOKEN" \
  -F "file=@./example.csv" \
  -F "password=optional"

# 2) 列出我的文件
curl http://localhost:3000/api/files \
  -H "Authorization: Bearer $TOKEN"

# 3) 删除
curl -X DELETE http://localhost:3000/api/files/1 \
  -H "Authorization: Bearer $TOKEN"

# 4) 公开下载（无密码时，8 位短码）
curl -OJ http://localhost:3000/api/s/Ab3xY9kQ/download
